//檔案合併作業
public object MergeChunks(dynamic param)
{
    //前端參數
    string fileName = param.fileName.ToString();
    int totalChunks = (int)param.totalChunks;
    long totalSize = (long)param.totalSize;
    
    // 最終檔案要存放的位置 (母目錄)
    string finalDir = @"\\10.1.90.127\plm_test\FILES\FM";
    //string finalDir = @"D:\EEPCore\EEPWebClient.Core\design\files\FM";
    //碎片的存放位置 (子目錄)
    string folderName = param.folderName.ToString();
    string chunkDir = Path.Combine(finalDir, folderName); 
    string finalPath = Path.Combine(finalDir, fileName);
    
     //重名則自動編號 EX: TEST(1) 、TEST(2)
    if (File.Exists(finalPath))
    {
        string extension = Path.GetExtension(fileName);
        string nameWithoutExt = Path.GetFileNameWithoutExtension(fileName);
        int count = 1;

        while (File.Exists(finalPath))
        {
            fileName = $"{nameWithoutExt}({count}){extension}";
            finalPath = Path.Combine(finalDir, fileName);
            count++;
        }
    }
    
    
    try
    {
        // 建立一個 4MB 的緩衝區，用來暫存讀取的資料，減少磁碟頻繁讀寫的壓力
        byte[] buffer = new byte[1024 * 1024 * 4]; 
        
        // 建立目標檔案串流 (finalPath)
        // FileMode.Create: 如果檔案已存在就覆蓋它
        // FileOptions.SequentialScan: 優化提示，告訴系統我們會「依序」寫入，加快速度
        using (var fs = new FileStream(finalPath, FileMode.Create, FileAccess.Write, FileShare.None, 4096, FileOptions.SequentialScan))
        {
            // 預先設定好檔案的總長度，這能讓作業系統一次分配好磁碟空間，避免寫入時產生磁碟碎片
            fs.SetLength(totalSize);
            // 按照編號依序處理每一個碎片
            for (int i = 0; i < totalChunks; i++)
            {
                // 拼接出碎片的完整路徑
                //string partPath = Path.Combine(chunkDir, $"{fileName}.part{i}");
                string partPath = Path.Combine(chunkDir, $"{param.fileName.ToString()}.part{i}");
                if (File.Exists(partPath))
                {
                    // 開啟碎片檔案進行讀取
                    using (var partStream = new FileStream(partPath, FileMode.Open, FileAccess.Read))
                    {
                        int bytesRead;
                        // 不斷從碎片檔案讀取資料放入 buffer，直到讀完為止
                        while ((bytesRead = partStream.Read(buffer, 0, buffer.Length)) > 0)
                        {
                            // 將 buffer 裡的資料寫入目標檔案 (fs)
                            fs.Write(buffer, 0, bytesRead);
                        }
                    }
                }
                else
                {
                    throw new Exception($"遺失分片: {fileName}.part{i}");
                }
            }
            fs.Flush(); // 離開 using 區塊時，fs 會自動關閉並釋放檔案鎖定
        }
        //合併成功後，直接砍掉整個碎片資料夾
        try 
        {
            if (Directory.Exists(chunkDir))
            {
                // 第二個參數設為 true，會連同裡面的所有檔案一起砍掉
                Directory.Delete(chunkDir, true);
            }
        } 
        catch (Exception ex) 
        {
            // Console.WriteLine("清理資料夾失敗: " + ex.Message);
        }
        return new { success = true, msg = "合併完成並已清理暫存區", actualFileName = fileName };
    }
    catch (Exception ex)
    {
        return new { success = false, msg = ex.Message };
    }
}
